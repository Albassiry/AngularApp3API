<?php
// Include database connection
require_once 'db.php';

$method = $_SERVER['REQUEST_METHOD'];

switch ($method) {
    case 'GET':
        // Fetch single book if ID is provided, otherwise fetch all books
        if (isset($_GET['id'])) {
            $stmt = $pdo->prepare("SELECT * FROM books WHERE id = ?");
            $stmt->execute([$_GET['id']]);
            $book = $stmt->fetch();
            echo json_encode($book ? $book : ["message" => "Book not found"]);
        } else {
            $stmt = $pdo->query("SELECT * FROM books ORDER BY id DESC");
            $books = $stmt->fetchAll();
            echo json_encode($books);
        }
        break;

    case 'POST':
        // Add a new book
        $data = json_decode(file_get_contents("php://input"), true);
        if (!empty($data['title']) && !empty($data['author'])) {
            $stmt = $pdo->prepare("INSERT INTO books (title, author, description) VALUES (?, ?, ?)");
            $stmt->execute([$data['title'], $data['author'], $data['description'] ?? '']);
            echo json_encode(["message" => "Book created successfully", "id" => $pdo->lastInsertId()]);
        } else {
            http_response_code(400);
            echo json_encode(["message" => "Incomplete data. Title and Author are required."]);
        }
        break;

    case 'PUT':
        // Update an existing book
        $data = json_decode(file_get_contents("php://input"), true);
        if (!empty($data['id']) && !empty($data['title']) && !empty($data['author'])) {
            $stmt = $pdo->prepare("UPDATE books SET title = ?, author = ?, description = ? WHERE id = ?");
            $stmt->execute([$data['title'], $data['author'], $data['description'] ?? '', $data['id']]);
            echo json_encode(["message" => "Book updated successfully"]);
        } else {
            http_response_code(400);
            echo json_encode(["message" => "Incomplete data for update."]);
        }
        break;

    case 'DELETE':
        // Delete a book by ID
        if (isset($_GET['id'])) {
            $stmt = $pdo->prepare("DELETE FROM books WHERE id = ?");
            $stmt->execute([$_GET['id']]);
            echo json_encode(["message" => "Book deleted successfully"]);
        } else {
            http_response_code(400);
            echo json_encode(["message" => "Missing book ID."]);
        }
        break;

    default:
        http_response_code(405);
        echo json_encode(["message" => "Method Not Allowed"]);
        break;
}
?>
